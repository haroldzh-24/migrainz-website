import type { Access, CollectionAfterReadHook, CollectionSlug, PayloadRequest, Where } from 'payload';
import { APIError } from 'payload';
import type { AccessState } from '@/cms/fields';

export const staff = ({ req }: { req: PayloadRequest }): boolean => Boolean(req.user?.collection === 'users');
export const publicStateWhere: Where = { or: [
  { 'accessControl.state': { in: ['public', 'redacted', 'patron'] } },
  { and: [{ 'accessControl.state': { exists: false } }, { listingVisibility: { equals: 'public' } }, { accessLevel: { equals: 'public' } }] },
] };

export const publicRecordWhere: Where = { and: [{ _status: { equals: 'published' } }, publicStateWhere] };
const legacyOrNewPublicState: Where = { or: [
  { 'accessControl.state': { equals: 'public' } },
  { and: [{ 'accessControl.state': { exists: false } }, { listingVisibility: { equals: 'public' } }, { accessLevel: { equals: 'public' } }] },
] };
export const publishedPublic: Where = { and: [{ _status: { equals: 'published' } }, legacyOrNewPublicState] };

type AccessMetadata = { state?: AccessState; displayTitle?: string | null; redactionLabel?: string | null; teaser?: string | null; showInListings?: boolean | null };
type PublicRecord = Record<string, unknown> & { accessControl?: AccessMetadata; accessLevel?: string; listingVisibility?: string };

export function accessState(record: PublicRecord): AccessState {
  const state = record.accessControl?.state;
  if (state && ['public', 'redacted', 'patron', 'hidden'].includes(state)) return state;
  if (record.listingVisibility === 'hidden') return 'hidden';
  if (record.accessLevel === 'patron') return 'patron';
  return 'public';
}

export function canAccessPatronContent(_req: PayloadRequest): boolean {
  return false;
}

function safeAccessMetadata(value: unknown): AccessMetadata {
  if (!value || typeof value !== 'object') return { state: 'public', showInListings: true };
  const fields = value as AccessMetadata;
  return {
    state: fields.state && ['public', 'redacted', 'patron', 'hidden'].includes(fields.state) ? fields.state : 'public',
    displayTitle: fields.displayTitle || undefined,
    redactionLabel: fields.redactionLabel || undefined,
    teaser: fields.teaser || undefined,
    showInListings: fields.showInListings !== false,
  };
}

const publicRecordFields = ['id', 'slug', 'project', 'comic'] as const;
const sectionSafeFields = ['id', 'blockType', 'accessControl'] as const;

function sanitizeSection(value: unknown): Record<string, unknown> | null {
  if (!value || typeof value !== 'object') return null;
  const section = value as PublicRecord;
  const control = safeAccessMetadata(section.accessControl);
  const state = accessState(section);
  if (state === 'hidden' || (state !== 'public' && control.showInListings === false)) return null;
  if (state === 'public') return { ...section, accessControl: control };
  const safe: Record<string, unknown> = {};
  for (const key of sectionSafeFields) if (key in section) safe[key] = section[key];
  safe.accessControl = { ...control, state };
  return safe;
}

export const redactPublicRecord: CollectionAfterReadHook = ({ doc, req }) => {
  if (req.user?.collection === 'users' || canAccessPatronContent(req)) return doc;
  const record = doc as PublicRecord;
  const state = accessState(record);
  const control = safeAccessMetadata(record.accessControl);
  if (state === 'hidden') return { id: record.id, accessControl: { state: 'hidden' } };
  if (state !== 'public') {
    const displayTitle = control.displayTitle || (state === 'patron' ? 'CLASSIFIED' : control.redactionLabel || 'REDACTED');
    const safe: PublicRecord = { accessControl: { ...control, state }, title: displayTitle, name: displayTitle };
    for (const key of publicRecordFields) if (key in record) {
      const value = record[key];
      safe[key] = key === 'project' || key === 'comic'
        ? value && typeof value === 'object' && 'id' in value ? { id: (value as { id: unknown }).id } : value
        : value;
    }
    return safe;
  }
  const sections = Array.isArray(record.sections) ? record.sections.map(sanitizeSection).filter(Boolean) : record.sections;
  return { ...record, accessControl: control, sections };
};

// Listing visibility deliberately does not grant content access. A future listing
// endpoint must explicitly project only approved metadata, never full documents.
export function publicContent(parent?: CollectionSlug, field = 'project', optional = false): Access {
  return async ({ req }) => {
    if (req.user?.collection === 'users') return true;
    if (!parent) return publicRecordWhere;
    const records = await req.payload.find({
      collection: parent, depth: 0, pagination: false, overrideAccess: false,
      where: publishedPublic, req,
    });
    const relation: Where = { [field]: { in: records.docs.map((doc) => doc.id) } };
    return { and: [publicRecordWhere, optional ? { or: [relation, { [field]: { exists: false } }] } : relation] };
  };
}

export const editorialAccess = (read: Access) => ({
  read, create: staff, update: staff, delete: staff, readVersions: staff,
});

// Media has its own access classification AND must be attached to published,
// accessible content. Hiding a chapter also hides files used only by that chapter.
const mediaRules = new WeakMap<PayloadRequest, Promise<Where>>();
export const mediaRead: Access = ({ req }) => {
  if (req.user?.collection === 'users') return true;
  // Many pages can reference media in one response. Share the rule only within
  // that request, never across requests or across publish/unpublish operations.
  let rule = mediaRules.get(req);
  if (!rule) { rule = buildMediaRule(req); mediaRules.set(req, rule); }
  return rule;
};
async function buildMediaRule(req: PayloadRequest): Promise<Where> {
  const ids = new Set<number | string>();
  const add = (value: unknown) => {
    if (typeof value === 'number' || typeof value === 'string') ids.add(value);
  };
  for (const collection of ['projects', 'factions', 'equipment', 'characters', 'comics', 'chapters', 'galleries', 'archive-items', 'project-updates'] as CollectionSlug[]) {
    const result = await req.payload.find({ collection, req, overrideAccess: false, depth: 0, pagination: false });
    for (const doc of result.docs) {
      const record = doc as unknown as Record<string, unknown>;
      add(record.heroImage); add(record.cover); add(record.emblem);
      for (const field of ['pages', 'images', 'files']) {
        const rows = record[field];
        if (Array.isArray(rows)) for (const row of rows) add(row.media);
      }
    }
  }
  return { and: [
    { _status: { equals: 'published' } },
    { or: [
      legacyOrNewPublicState,
    ] },
    { id: { in: [...ids] } },
  ] };
}

export async function preventReferencedDelete(req: PayloadRequest, collection: CollectionSlug, id: number | string) {
  const references: Partial<Record<CollectionSlug, [CollectionSlug, string][]>> = {
    projects: [['factions', 'project'], ['equipment', 'project'], ['characters', 'project'], ['comics', 'project'], ['project-updates', 'project'], ['tracker-items', 'project'], ['galleries', 'project'], ['archive-items', 'project']],
    factions: [['characters', 'primaryFaction'], ['characters', 'affiliations'], ['equipment', 'faction']],
    comics: [['chapters', 'comic']],
    chapters: [['characters', 'relatedChapters']],
    media: [['factions', 'emblem'], ['equipment', 'images.media'], ['projects', 'heroImage'], ['characters', 'images.media'], ['comics', 'cover'], ['chapters', 'pages.media'], ['galleries', 'images.media'], ['archive-items', 'files.media'], ['project-updates', 'images.media']],
    galleries: [['projects', 'galleries']],
    tags: [['projects', 'tags'], ['characters', 'tags'], ['galleries', 'tags'], ['archive-items', 'tags']],
    categories: [['projects', 'categories'], ['archive-items', 'category']],
  };
  for (const [target, field] of references[collection] ?? []) {
    const result = await req.payload.find({ collection: target, req, overrideAccess: true, depth: 0,
      limit: 1, where: { [field]: { equals: id } } });
    if (result.totalDocs) throw new APIError(`Remove references in ${target} before deleting this ${collection} record.`, 409);
    if (req.payload.collections[target].config.versions) {
      const drafts = await req.payload.find({ collection: target, req, overrideAccess: true, depth: 0,
        draft: true, limit: 1, where: { [field]: { equals: id } } });
      if (drafts.totalDocs) throw new APIError(`Remove draft references in ${target} before deleting this ${collection} record.`, 409);
    }
  }
}
